       public ActionResult view_person()
        {
            if (Session["userid"] == null)
            {
                Session["alarmclass"] = "alarm_box";
                Session["alarmText_title"] = "خطا در ثبت اطلاعات";
                Session["btn_class"] = "btn-Pcustome-alarm";
                Session["td_class"] = "red";
                Session["alarmText"] = "شما مدت زیادی است که از سیستم استفاده نکرده‌اید. لطفا مجددا وارد شوید.";
                Session["alarmpic"] = "/Content/images/Cancel-128.png";
                return RedirectToAction("../home/LogInFS");
            }

            HomeController HC = new HomeController();
            set_alarm();
            HC.set_active_menu("MA8", "MA81");
            user_not_access test = new user_not_access();
            string[] str = test.initial_access(int.Parse(Session["userid"].ToString()), 5);
            int j = 0;
            while (str[j] != null)
            {
                ViewData[str[j]] = ",@disabled";
                j++;
            }
            set_personcombo();
            String soccer_id = Request["soccer_id"];
            fs_period fsp = new fs_period();
            if (string.IsNullOrEmpty(System.Web.HttpContext.Current.Session["period_id"] as String))
                Session["period_id"] = fsp.last_fsperiod(long.Parse(Session["fschool_id"].ToString()));
            conect_db.CommandTimeout = 1200;
            var result_soccer = conect_db.sp_fs_soccer_list(long.Parse(Session["fschool_id"].ToString())
                                                                , long.Parse(Session["userid"].ToString())
                                                                , long.Parse(Session["period_id"].ToString())
                                                                , "");

            List<SelectListItem> soccer = new List<SelectListItem>();
            foreach (var item in result_soccer)
            {
                SelectListItem temp_soccer = new SelectListItem();
                temp_soccer.Text = item.soccer_name;
                temp_soccer.Value = item.user_id_FK.ToString();
                soccer.Add(temp_soccer);
            }

            ViewBag.soccer = soccer;

            long? flag = 0;
            if (soccer.Count == 0)
            {
                return View("view_person");
            }

            if ((soccer_id == "" || soccer_id == null))
            {
                flag = long.Parse(soccer.First().Value);
                soccer_id = flag.ToString();
            }

            var model = set_personinfo(long.Parse(soccer_id));
            person rec = new person();

            if (model.Count() > 0)
            {
                rec = model.First();
                Session["selectsoccer"] = rec.user_id_FK.ToString();
                if (flag != 0)
                {
                    return View("view_person", rec);
                }
                else
                {
                    return Json(new { model = rec }, JsonRequestBehavior.AllowGet);
                }

            }
            else
            {
                return View("view_person");
            }
        }
